        Application Structure Plan:

        The SPA is designed as an interactive dashboard to give faculty a quick, actionable overview. The structure flows from a high-level summary to specific details, encouraging exploration.

        1. KPIs: Instant summary of the program's status.

        2. Interactive Schedule: The core of the app. A filterable card view is more user-friendly than a table, allowing users to immediately find opportunities ("Needs Presenter"). This is the primary user task.

        3. Data Insights: Charts provide context and a deeper understanding of the program's composition, answering questions like "Which partners are most active?" and "What kinds of topics are we covering?".

        4. Volunteer Hub: A clear call-to-action section that explains the process and showcases existing volunteers to encourage participation.

        This thematic, task-oriented structure is chosen over a simple table to make the information more engaging and to guide users toward the key goal: filling open presentation slots.

 

        Visualization & Content Choices:

        - KPIs -> Goal: Inform -> Viz: Large Stats (HTML/CSS) -> Interaction: None -> Justification: Provides a quick, high-impact summary.

        - Schedule -> Goal: Organize/Inform -> Viz: Interactive Cards (HTML/CSS) -> Interaction: Filtering by status/partner -> Justification: Allows users to easily find relevant information, especially open slots.

        - Sessions by Partner -> Goal: Compare -> Viz: Bar Chart (Chart.js/Canvas) -> Interaction: Tooltips -> Justification: Best for comparing counts across distinct categories.

        - Topic Sources -> Goal: Compare (Composition) -> Viz: Donut Chart (Chart.js/Canvas) -> Interaction: Tooltips -> Justification: Excellent for showing parts of a whole.

        - Volunteer Process -> Goal: Organize -> Viz: Flowchart Diagram (HTML/CSS) -> Interaction: None -> Justification: Clearly illustrates a multi-step process without complex visuals.

        - CONFIRMATION: NO SVG graphics used. NO Mermaid JS used.

    -->

    <script>

        // =================================================================

        // --- IMPORTANT: PASTE YOUR GOOGLE SHEET URLs IN THE LINES BELOW ---

        // =================================================================

        const SCHEDULE_DATA_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vSMxqVM8Fgks5K4SOWuntC4DsjQbPAIuS-nlOs8i0a-YIKLjkNU8hmz2C02ZVLWW0z_2gF1cWNNIEat/pub?gid=0&single=true&output=csv';

        const VOLUNTEER_DATA_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vSMxqVM8Fgks5K4SOWuntC4DsjQbPAIuS-nlOs8i0a-YIKLjkNU8hmz2C02ZVLWW0z_2gF1cWNNIEat/pub?gid=896484327&single=true&output=csv';

        // =================================================================

 

        document.addEventListener('DOMContentLoaded', () => {

            const scheduleGrid = document.getElementById('schedule-grid');

            const filtersContainer = document.getElementById('filters');

            const volunteerList = document.getElementById('volunteer-list');

            let scheduleData = [];

 

            const csvToJSON = (csv) => {

                const lines = csv.split('\n');

                const result = [];

                const headers = lines[0].split(',').map(h => h.trim().replace(/"/g, ''));

                for (let i = 1; i < lines.length; i++) {

                    if (!lines[i]) continue;

                    const obj = {};

                    const currentline = lines[i].split(',');

                    for (let j = 0; j < headers.length; j++) {

                        obj[headers[j]] = currentline[j] ? currentline[j].trim().replace(/"/g, '') : '';

                    }

                    result.push(obj);

                }

                return result;

            };

 

            const renderSchedule = (filter = 'all') => {

                scheduleGrid.innerHTML = '';

                const filteredData = scheduleData.filter(item => {

                    if (filter === 'all') return true;

                    return item.status === filter;

                });

 

                if (filteredData.length === 0) {

                    scheduleGrid.innerHTML = `<p class="text-stone-500 md:col-span-2 text-center py-8">No sessions match the current filter.</p>`;

                    return;

                }

 

                filteredData.forEach(item => {

                    const card = document.createElement('div');

                    const statusClass = item.status === 'needs' ? 'border-asu-gold' : 'border-asu-maroon';

                    const statusText = item.status === 'needs' ? 'Needs Presenter' : 'Confirmed';

                    const statusTextColor = item.status === 'needs' ? 'text-amber-600' : 'text-asu-maroon';

                    const presenterInfo = item.presenter ? `<p class="text-sm text-stone-500 mt-1"><strong>Presenter:</strong> ${item.presenter}</p>` : `<p class="text-sm font-bold ${statusTextColor} mt-1">This session needs a presenter.</p>`;

 

                    card.className = `bg-stone-50 p-5 rounded-lg border-l-4 ${statusClass} transition hover:shadow-lg`;

                    card.innerHTML = `

                        <div class="flex justify-between items-start">

                            <div>

                                <p class="text-sm font-semibold text-asu-maroon">${item.partner}</p>

                                <h4 class="text-lg font-bold text-stone-800 mt-1">${item.topic}</h4>

                            </div>

                            <span class="text-xs font-bold ${statusTextColor} bg-opacity-20 ${item.status === 'needs' ? 'bg-amber-100' : 'bg-red-50'} py-1 px-3 rounded-full">${statusText}</span>

                        </div>

                        <div class="mt-4 border-t border-stone-200 pt-3">

                            <p class="text-sm text-stone-500"><strong>Date:</strong> ${item.date}</p>

                            ${presenterInfo}

                            ${item.notes ? `<p class="text-sm text-stone-500 mt-1"><strong>Notes:</strong> ${item.notes}</p>` : ''}

                        </div>

                    `;

                    scheduleGrid.appendChild(card);

                });

            };

 

            const renderVolunteers = (data) => {

                volunteerList.innerHTML = '';

                if (!data || data.length === 0) {

                    volunteerList.innerHTML = `<p class="text-stone-200 md:col-span-2 text-center py-4">No volunteers have signed up yet.</p>`;

                    return;

                }

                data.forEach(volunteer => {

                    const card = document.createElement('div');

                    card.className = 'bg-white/10 p-4 rounded-lg';

                    card.innerHTML = `

                        <p class="font-bold text-lg text-white">${volunteer.name}</p>

                        <p class="text-stone-300">Expertise: ${volunteer.topic}</p>

                    `;

                    volunteerList.appendChild(card);

                });

            };

 

            filtersContainer.addEventListener('click', (e) => {

                if (e.target.tagName === 'BUTTON') {

                    document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));

                    e.target.classList.add('active');

                    const filter = e.target.dataset.filter;

                    renderSchedule(filter);

                }

            });

 

            const updateKPIs = (data) => {

                const partners = new Set(data.map(s => s.partner));

                document.getElementById('kpi-partnerships').textContent = partners.size;

                document.getElementById('kpi-sessions').textContent = data.length;

                const confirmedCount = data.filter(s => s.status === 'confirmed').length;

                const neededCount = data.filter(s => s.status === 'needs').length;

                document.getElementById('kpi-confirmed').textContent = confirmedCount;

                document.getElementById('kpi-needed').textContent = neededCount;

            };

 

            const createCharts = (data) => {

                const tooltipTitleCallback = (tooltipItems) => {

                    const item = tooltipItems[0];

                    let label = item.chart.data.labels[item.dataIndex];

                    return Array.isArray(label) ? label.join(' ') : label;

                };

                const chartOptions = { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'bottom', labels: { font: { family: 'Inter' } } }, tooltip: { callbacks: { title: tooltipTitleCallback } } } };

               

                const partnerCounts = data.reduce((acc, session) => {

                    acc[session.partner] = (acc[session.partner] || 0) + 1;

                    return acc;

                }, {});

 

                new Chart(document.getElementById('sessionsByPartnerChart').getContext('2d'), {

                    type: 'bar',

                    data: { labels: Object.keys(partnerCounts), datasets: [{ label: 'Number of Sessions', data: Object.values(partnerCounts), backgroundColor: ['#8C1D40', '#00A3E0', '#747474'] }] },

                    options: { ...chartOptions, scales: { y: { beginAtZero: true, ticks: { stepSize: 1 } } } }

                });

 

                const topicSources = { 'District Request': 1, 'Course Alignment': 2, 'Faculty Volunteer': 1 };

                new Chart(document.getElementById('topicSourceChart').getContext('2d'), {

                    type: 'doughnut',

                    data: { labels: Object.keys(topicSources), datasets: [{ data: Object.values(topicSources), backgroundColor: ['#FFC627', '#8C1D40', '#00A3E0'], borderColor: '#FFFFFF', borderWidth: 2 }] },

                    options: chartOptions

                });

            };

 

            async function loadScheduleData() {

                if (SCHEDULE_DATA_URL.startsWith('PASTE_YOUR')) {

                    scheduleGrid.innerHTML = `<p class="text-amber-700 font-semibold md:col-span-2 text-center py-8">Configuration needed: Please paste your Schedule Google Sheet URL into the script.</p>`;

                    return;

                }

                try {

                    const response = await fetch(SCHEDULE_DATA_URL);

                    if (!response.ok) throw new Error('Network response was not ok.');

                    const csvData = await response.text();

                    scheduleData = csvToJSON(csvData);

                    renderSchedule();

                    updateKPIs(scheduleData);

                    createCharts(scheduleData);

                } catch (error) {

                    console.error('Failed to load schedule data:', error);

                    scheduleGrid.innerHTML = `<p class="text-red-500 md:col-span-2 text-center py-8">Error: Could not load schedule data. Please check the URL and sharing settings.</p>`;

                }

            }

 

            async function loadVolunteerData() {

                if (VOLUNTEER_DATA_URL.startsWith('PASTE_YOUR')) {

                    volunteerList.innerHTML = `<p class="text-amber-300 font-semibold md:col-span-2 text-center py-4">Configuration needed: Please paste your Volunteers Google Sheet URL into the script.</p>`;

                    return;

                }

                try {

                    const response = await fetch(VOLUNTEER_DATA_URL);

                    if (!response.ok) throw new Error('Network response was not ok.');

                    const csvData = await response.text();

                    const volunteerData = csvToJSON(csvData);

                    renderVolunteers(volunteerData);

                } catch (error) {

                    console.error('Failed to load volunteer data:', error);

                    volunteerList.innerHTML = `<p class="text-red-300 md:col-span-2 text-center py-4">Error: Could not load volunteer data. Please check the URL and sharing settings.</p>`;

                }

            }

 

            loadScheduleData();

            loadVolunteerData();

        });

    </script>
